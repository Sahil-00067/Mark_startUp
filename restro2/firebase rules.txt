rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    match /restaurants/{restaurantId} {

      // Public menu + profile - world readable (QR customers)
      allow read: if true;

      // Top-level key whitelist - must match what the app saves
      allow create, update: if request.resource.data.keys().hasOnly(['restaurant', 'categories', 'items', 'activityLog', 'nextId', 'nextOrderId', '_updatedAt'])
        && request.resource.data.restaurant is map
        && request.resource.data.items is list
        && request.resource.data.items.size() <= 300;

      // No client may delete the restaurant profile/menu doc
      allow delete: if false;

      // ---------------- Orders ----------------
      match /orders/{orderId} {
        // Live tracking (customer) + kitchen display
        allow read: if true;

        // New orders must start pending + unpaid with a sane item list
        // (blocks forged already-paid orders / free food)
        allow create: if request.resource.data.keys().hasAll(['id', 'orderNum', 'table', 'customer', 'phone', 'items', 'status', 'payment', 'ts'])
          && request.resource.data.status == 'pending'
          && request.resource.data.payment == 'unpaid'
          && request.resource.data.items is list
          && request.resource.data.items.size() > 0
          && request.resource.data.items.size() <= 50;

        // Bookkeeping fields ONLY - single line, do not break it.
        allow update: if request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'payment', 'payMethod', '_new', 'preparingAt', 'preparedAt', 'cancelReason', 'rating', 'ratedAt']);

        // Admin "Reset All" + removing test orders need deletes
        allow delete: if true;
      }

      // -------- Waiter calls / checkout requests --------
      match /waiterCalls/{callId} {
        allow read: if true;

        // Customers can call as Guest - minimal required keys
        allow create: if request.resource.data.keys().hasAll(['id', 'table', 'status', 'ts']);

        // Status + checkout bookkeeping - single line, do not break it
        allow update: if request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'resolvedAt', 'resolvedMethod', 'resolvedBy', 'note']);

        // Staff board cleanup
        allow delete: if true;
      }

      // ---------------- Photos (base64) ----------------
      match /photos/{photoId} {
        allow read, write: if true;
      }
    }
  }
}

WHY THE OLD RULES BROKE THE APP (for reference):
- Orders update whitelist was missing preparingAt / preparedAt /
  cancelReason / rating / ratedAt. The kitchen wrote status TOGETHER
  with timestamps in one update, so Firestore rejected the WHOLE
  write - customer phones never saw Cooking/Ready/Served.
- Main doc only allowed exact top-level keys; one stray key made the
  whole save fail, so menu edits stopped syncing (looked like items
  "disappeared" - they were stale snapshots, cloud was fine).
The app code is already fixed for both; these rules complete it.

WHAT THE RULES PROTECT (kept from your hardened version):
- New orders must be pending + unpaid (no forged paid orders)
- No price/quantity rewriting after creation
- No deleting the restaurant/menu doc
- No credentials ever leave the device
